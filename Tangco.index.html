    <script>

        const calcServiceType = document.getElementById('calcServiceType');

        const calcSqft = document.getElementById('calcSqft');

        const sqftValue = document.getElementById('sqftValue');

        const totalPrice = document.getElementById('totalPrice');

        const calcAddons = document.querySelectorAll('.calc-addon');


        function updatePrice() {

            const sqft = parseInt(calcSqft.value);

            sqftValue.innerText = sqft + " 呎";

            

            let base = parseInt(calcServiceType.value);

            let sqftCost = (sqft - 300) * 2.5;

            if(sqftCost < 0) sqftCost = 0;


            let addonTotal = 0;

            calcAddons.forEach(addon => {

                if(addon.checked) addonTotal += parseInt(addon.value);

            });


            let final = base + sqftCost + addonTotal;

            totalPrice.innerText = final.toLocaleString();

        }


        calcServiceType.addEventListener('change', updatePrice);

        calcSqft.addEventListener('input', updatePrice);

        calcAddons.forEach(addon => addon.addEventListener('change', updatePrice));

        updatePrice();


        function getBookings() {

            let data = localStorage.getItem('tangco_bookings');

            return data ? JSON.parse(data) : [];

        }


        function saveBooking(booking) {

            let bookings = getBookings();

            bookings.unshift(booking);

            localStorage.setItem('tangco_bookings', JSON.stringify(bookings));

            renderBookings();

        }


        function renderBookings() {

            let bookings = getBookings();

            let tbody = document.getElementById('bookingsTableBody');

            let noMsg = document.getElementById('noBookingsMsg');


            tbody.innerHTML = '';

            if(bookings.length === 0) {

                noMsg.style.display = 'block';

                return;

            }

            noMsg.style.display = 'none';


            bookings.forEach((b, index) => {

                let tr = document.createElement('tr');

                tr.className = "hover:bg-slate-50";

                tr.innerHTML = `

                    <td class="p-4 font-bold text-[#0B2545]">${b.name}</td>

                    <td class="p-4">${b.phone}</td>

                    <td class="p-4"><span class="bg-blue-100 text-blue-800 text-xs px-2.5 py-1 rounded-full font-bold">${b.service}</span></td>

                    <td class="p-4">${b.date}</td>

                    <td class="p-4 text-slate-600 text-xs">${b.notes || '無'}</td>

                    <td class="p-4"><span class="bg-amber-100 text-amber-800 text-xs px-2.5 py-1 rounded-full font-bold">待確認</span></td>

                `;

                tbody.appendChild(tr);

            });

        }
